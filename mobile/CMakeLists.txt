cmake_minimum_required(VERSION 3.22)
project(maglava_mobile C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
get_filename_component(V4 "${CMAKE_CURRENT_SOURCE_DIR}/../v4" ABSOLUTE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
file(GLOB LEVEL_JSON CONFIGURE_DEPENDS "${V4}/levels/*.json")
set(GENERATED "${CMAKE_CURRENT_BINARY_DIR}/generated/levels_gen.h")
add_custom_command(OUTPUT "${GENERATED}"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/generate_levels.py" "${GENERATED}"
    DEPENDS ${LEVEL_JSON} "${V4}/tools/gen_levels.py" tools/generate_levels.py VERBATIM)
add_library(maglava_core STATIC shared/mobile_core.c shared/presentation.c shared/levels.c "${V4}/src/sim.c" "${GENERATED}")
set_target_properties(maglava_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(maglava_core PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" PUBLIC shared "${V4}/src")
target_compile_options(maglava_core PRIVATE -Wall -Wextra)
target_link_libraries(maglava_core PUBLIC m)
if(NOT ANDROID)
    enable_testing()
    add_executable(presentation_test tests/test_presentation.c)
    target_link_libraries(presentation_test PRIVATE maglava_core)
    add_test(NAME shared_presentation COMMAND presentation_test)
    add_executable(mobile_core_test tests/test_core.c)
    target_link_libraries(mobile_core_test PRIVATE maglava_core)
    add_test(NAME mobile_frame_contract COMMAND mobile_core_test)
    add_executable(sim_test "${V4}/tests/test_sim.c")
    target_link_libraries(sim_test PRIVATE maglava_core)
    add_test(NAME gameplay_regression COMMAND sim_test)
    add_executable(campaign_audit "${V4}/tools/campaign_audit.c")
    target_link_libraries(campaign_audit PRIVATE maglava_core)
    add_test(NAME campaign COMMAND campaign_audit --strict)
    add_test(NAME relaxed_campaign COMMAND campaign_audit --strict --new-relaxed)
    add_test(NAME subspace COMMAND campaign_audit --strict --deathless --new-relaxed --level-key level-6b)
endif()
